FLOW IN CLONING THE RETAIL BIKE STOCK TRACKING WEBSITE
# 1. Clone the repository
git clone <your-repo-url>
cd ELEC3_Montalan

# 2. Install PHP dependencies
composer install

# 3. Install packages
npm install

# 4. Configure environment
copy .env.example 

# 5. Generate APP_KEY
php artisan key:generate

# 6. Uncomment database connection inside .env and change accordingly
DB_CONNECTION=mysql
DB_HOST=127.0.0.1
DB_PORT=3306
DB_DATABASE=montalandb
DB_USERNAME=root
DB_PASSWORD=


# 7. Run migrations and seeders
php artisan migrate:fresh --seed 




ADDITIONAL:
php artisan storage:link

FOR ANY ERRORS:
php artisan optimize:clear 
php artisan cache:clear
# 8. Start the application
composer run dev (all in one command for npm run build, php artisan serve, etc)
# App runs at http://127.0.0.1:8000
